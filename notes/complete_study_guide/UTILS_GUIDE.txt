═══════════════════════════════════════════════════════════════════
✅ WHAT utils.c SHOULD CONTAIN
═══════════════════════════════════════════════════════════════════

Only minishell-specific utility functions that don't belong
in any other module!

Example functions that SHOULD be in utils.c:
--------------------------------------------

1. Array length counter:
   int array_len(char **arr)
   {
       int i = 0;
       if (!arr)
           return (0);
       while (arr[i])
           i++;
       return (i);
   }

2. Safe string comparison (handles NULL):
   int safe_strcmp(const char *s1, const char *s2)
   {
       if (!s1 && !s2)
           return (0);
       if (!s1)
           return (-1);
       if (!s2)
           return (1);
       return (ft_strncmp(s1, s2, ft_strlen(s1) + 1));
   }

3. Check if string is numeric:
   int is_numeric(char *str)
   {
       int i = 0;
       if (!str || !str[0])
           return (0);
       if (str[0] == '-' || str[0] == '+')
           i++;
       while (str[i])
       {
           if (!ft_isdigit(str[i]))
               return (0);
           i++;
       }
       return (1);
   }

4. Check if string is valid identifier (for export/unset):
   int is_valid_identifier(char *str)
   {
       int i = 0;
       if (!str || !str[0])
           return (0);
       if (!ft_isalpha(str[0]) && str[0] != '_')
           return (0);
       i = 1;
       while (str[i])
       {
           if (!ft_isalnum(str[i]) && str[i] != '_')
               return (0);
           i++;
       }
       return (1);
   }

5. Find character not inside quotes:
   int find_unquoted_char(char *str, char c)
   {
       int i = 0;
       int in_squote = 0;
       int in_dquote = 0;
       
       while (str[i])
       {
           if (str[i] == '\'' && !in_dquote)
               in_squote = !in_squote;
           else if (str[i] == '"' && !in_squote)
               in_dquote = !in_dquote;
           else if (str[i] == c && !in_squote && !in_dquote)
               return (i);
           i++;
       }
       return (-1);
   }

═══════════════════════════════════════════════════════════════════
KEY PRINCIPLE
═══════════════════════════════════════════════════════════════════

If it's a general-purpose function that could be used in ANY C program:
  → It should be in libft (or already is!)

If it's specific to minishell's logic:
  → It should be in utils.c
